### 1. 改写 `apps/code/src/index.ts`

用 `getConfig` + `ProviderConfigSchema` 替代 `bootstrap` + `buildLLMConfigFromEnv`。

关键改动：
```typescript
import { getConfig, type ConfigSource } from "@n0n/config";
import { ProviderConfigSchema, type ProviderConfig, createLLMClient, createResponsesClient } from "@n0n/llm";
import { z } from "zod";

// 1. 定义 settings schema
const codeConfigSchema = z.object({
  settings: z.object({
    strip_hint: z.boolean().default(true),
    notify_sound: z.boolean().default(false),
    notify_sound_path: z.string().default(""),
    blocked_commands: z.array(z.string()).default([]),
    llm: ProviderConfigSchema,
    editor: ProviderConfigSchema,
  }),
});

// 2. 默认配置（app 内联）
const DEFAULT_TOML = `...`;

// 3. 加载全局 + 项目 TOML
const sources: ConfigSource[] = [
  { name: "默认", content: DEFAULT_TOML },
  { name: "全局", content: readGlobalToml() },
];
if (projectTomlExists) sources.push({ name: "项目", content: readProjectToml() });

// 4. 解析
const envPool: Record<string, string> = {};
for (const [k, v] of Object.entries(process.env)) if (v !== undefined) envPool[k] = v;
const result = getConfig(codeConfigSchema, sources, envPool);
if (!result.success) { /* 报错退出 */ }

// 5. 构造组件
const { llm, editor } = result.data.settings;
const llmConfig = { providerConfig: llm };
const editBackendType = editor.edit_backend;  // 直接读，不需要 resolveEditBackend
```
